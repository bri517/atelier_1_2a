        Si tab[i] <- prixMin 
            Alors prixMin <- tab[i]
        Fin si

            Alors prixMax <- tab[i]
        Fin si 
        moyenne <- moyenne + tab[i]
    Fin pour

Fin tableau 
                    
#include <stdio.h>

int main(){
    float tab[10];
    float prixMax,prixMin,prixMoyen;
    int i;
    for(i=0;i<10;i++){
        printf("Donner un prix :");
        scanf("%f",&tab[i]);
    }
    prixMax =tab[0];
    prixMin = tab[0];
    prixMoyen = 0;
    for(i=0;i<10;i++){
        if(tab[i]>prixMax){
            prixMax = tab[i];
        }
        if (tab[i]<prixMin){
            prixMin = tab[i];
        }
        prixMoyen = prixMoyen + tab[i];
    }
    prixMoyen = prixMoyen/10;
    printf ("Le prix moyen est : %f\n",prixMoyen);
    printf ("Prix Max :%f\n et prix Min : %f",prixMax,prixMin);
    return 0;
}

    fd = fopen(nomD, "w"); 
    while ( ! feof(fs)){
        car = fgetc(fs); 
        fputc(car, fd);

    fclose(fs); 
    fclose(fd);
    return 0; 
}
